# Optional lab guard. The caller supplies a fixed digest-bound kernel prefix.
# Requests name one RTC minute. A 120-tick watchdog reset must outlast it.
# RTC advancement and the firmware/FreeBSD clock convention require physical
# validation before this can guard an armed experiment.
insmod datehook
function qotom_watchdog_window {
    set wd_window_prefix=watchdog-test
    if [ -n "$2" ]; then
        set wd_window_prefix=$2
    fi
    set wd_window="${wd_window_prefix}-${YEAR}-${MONTH}-${DAY}-${HOUR}-${MINUTE}"
    if [ "$YEAR" -ge 2026 -a "$YEAR" -le 2099 -a "$MONTH" -ge 1 -a "$MONTH" -le 12 -a "$DAY" -ge 1 -a "$DAY" -le 31 -a "$HOUR" -ge 0 -a "$HOUR" -le 23 -a "$MINUTE" -ge 0 -a "$MINUTE" -le 59 ]; then
        # Re-sample to reject rollover while constructing the first string.
        if [ "$1" = "$wd_window" -a "$wd_window" = "${wd_window_prefix}-${YEAR}-${MONTH}-${DAY}-${HOUR}-${MINUTE}" ]; then
            true
        else
            false
        fi
    else
        false
    fi
}

function qotom_watchdog_stop {
    # Called only after the exact pre-arm state below has been accepted.
    outw 0x468 0x800
    outw 0x472 4 0x3ff
    write_dword 0xfed03008 0x10 0x10
    inw -v wd_count 0x468
    read_dword -v wd_pmc 0xfed03008
    if [ "$wd_count" = "800" -a "$wd_pmc" = "10" ]; then
        set wd_armed=0
        true
    else
        echo 'LEANOS-LAB/1 WATCHDOG-STOP-FAILED'
        false
    fi
}

function qotom_watchdog_arm {
    set wd_armed=0
    set wd_host=unknown
    set wd_lpc=unknown
    set wd_acpi=unknown
    set wd_pbase=unknown
    setpci -s 00:00.0 -v wd_host 0.l
    setpci -s 00:1f.0 -v wd_lpc 0.l
    setpci -s 00:1f.0 -v wd_acpi 40.l
    setpci -s 00:1f.0 -v wd_pbase 44.l
    if [ "$wd_host" = "f008086" -a "$wd_lpc" = "f1c8086" -a "$wd_acpi" = "403" -a "$wd_pbase" = "fed03002" ]; then
        read_dword -v wd_pmc 0xfed03008
        inl -v wd_smi 0x430
        inw -v wd_count 0x468
        inw -v wd_timer 0x472
        echo "LEANOS-LAB/1 WATCHDOG-STATE pmc=$wd_pmc smi=$wd_smi count=$wd_count timer=$wd_timer"
        set wd_smi_ok=0
        if [ "$wd_smi" = "20033" -o "$wd_smi" = "20073" ]; then
            set wd_smi_ok=1
        fi
        if [ "$wd_pmc" = "10" -a "$wd_count" = "800" -a "$wd_timer" = "4" -a "$wd_smi_ok" = "1" ]; then
            # TCO SMI is already disabled. Preserve unrelated PMC/timer bits.
            write_dword 0xfed03008 0 0x10
            outw 0x464 8
            outw 0x466 2
            outw 0x466 4
            outw 0x472 120 0x3ff
            outw 0x460 1
            read_dword -v wd_pmc 0xfed03008
            inw -v wd_timer 0x472
            if [ "$wd_pmc" = "0" -a "$wd_timer" = "78" ]; then
                # Initial control was exactly HALT, so no preserve bit is set.
                outw 0x468 0
                inw -v wd_count 0x468
                if [ "$wd_count" = "0" ]; then
                    set wd_armed=1
                    echo 'LEANOS-LAB/1 WATCHDOG-ARMED ticks=120'
                fi
            fi
            if [ "$wd_armed" != "1" ]; then
                qotom_watchdog_stop
            fi
        fi
    fi
    if [ "$wd_armed" = "1" ]; then
        true
    else
        echo 'LEANOS-LAB/1 WATCHDOG-ARM-REJECTED'
        false
    fi
}
